// --- LOGIC HELPER ---
const calculateKRProgress = (target, actual, type) => {
  if (target === 0) return actual === 0 ? 100 : 0; 
  if (type === 'higher_better') {
    return Math.min(Math.round((actual / target) * 100), 100);
  } else {
    // Logic untuk NPL (Makin rendah makin bagus)
    if (actual <= target) return 100;
    const over = actual - target;
    const penalty = (over / target) * 100;
    return Math.max(Math.round(100 - penalty), 0);
  }
};

const getTrafficLight = (progress) => {
  if (progress >= 100) return { color: 'bg-emerald-500', bg: 'bg-emerald-100', text: 'text-emerald-700 dark:text-emerald-400', label: 'On Track' };
  if (progress >= 70) return { color: 'bg-amber-500', bg: 'bg-amber-100', text: 'text-amber-700 dark:text-amber-400', label: 'Warning' };
  return { color: 'bg-red-500', bg: 'bg-red-100', text: 'text-red-700', label: 'Off Track' };
};

              </div>
              
              {/* Form Tampilan Berbeda Berdasarkan Mode */}
              {modalObj.mode === 'edit-obj' ? (
                <div>
                  <label className="block text-sm font-semibold text-slate-700 dark:text-gray-200 mb-1.5">Teks Objective Utama</label>
                  <textarea 
                    rows="3" required value={formData.title} onChange={(e) => setFormData({...formData, title: e.target.value})} 
                    className="w-full border border-slate-300 dark:border-gray-600 rounded-lg p-2.5 outline-none text-sm focus:ring-2 focus:ring-blue-500" 
                    placeholder="Masukkan tujuan utama kuartal ini..." 
                  />
                </div>
              ) : (
                <>
                  <div>
                    <label className="block text-sm font-semibold text-slate-700 dark:text-gray-200 mb-1.5">Deskripsi Key Result</label>
                    <input 
                      type="text" required value={formData.title} onChange={(e) => setFormData({...formData, title: e.target.value})} 
                      className="w-full border border-slate-300 dark:border-gray-600 rounded-lg p-2.5 outline-none text-sm focus:ring-2 focus:ring-blue-500" 
                      placeholder="Contoh: Growth kredit mencapai 15%..." 
                    />
                  </div>

                  <div className="grid grid-cols-2 gap-4">
                    <div>
                      <label className="block text-sm font-semibold text-slate-700 dark:text-gray-200 mb-1.5">Target</label>
                      <input 
                        type="number" required value={formData.target} onChange={(e) => setFormData({...formData, target: e.target.value})} 
                        className="w-full border border-slate-300 dark:border-gray-600 rounded-lg p-2.5 outline-none text-sm focus:ring-2 focus:ring-blue-500" 
                        placeholder="Angka Target" step="any"
                      />
                    </div>
                    <div>
                      <label className="block text-sm font-semibold text-slate-700 dark:text-gray-200 mb-1.5">Realisasi (Aktual)</label>
                      <input 
                        type="number" required value={formData.actual} onChange={(e) => setFormData({...formData, actual: e.target.value})} 
                        className="w-full border border-slate-300 dark:border-gray-600 rounded-lg p-2.5 outline-none text-sm focus:ring-2 focus:ring-blue-500" 
                        placeholder="Angka Aktual" step="any"
                      />
                    </div>
                  </div>

                  <div className="grid grid-cols-2 gap-4">
                    <div>
                      <label className="block text-sm font-semibold text-slate-700 dark:text-gray-200 mb-1.5">Satuan</label>
                      <select 
                        value={formData.unit} onChange={(e) => setFormData({...formData, unit: e.target.value})} 
                        className="w-full border border-slate-300 dark:border-gray-600 rounded-lg p-2.5 outline-none text-sm focus:ring-2 focus:ring-blue-500 bg-white dark:bg-[#18181B]"
                      >
                        <option value="">(Pilih Satuan)</option>
                        <option value="%">Persen (%)</option>
                        <option value="Rp">Rupiah (Rp)</option>
                        <option value="Nasabah">Nasabah</option>
                        <option value="Dokumen">Dokumen</option>
                      </select>
                    </div>
                    <div>
                      <label className="block text-sm font-semibold text-slate-700 dark:text-gray-200 mb-1.5">Tipe Evaluasi</label>
                      <select 
                        value={formData.type} onChange={(e) => setFormData({...formData, type: e.target.value})} 
                        className="w-full border border-slate-300 dark:border-gray-600 rounded-lg p-2.5 outline-none text-sm focus:ring-2 focus:ring-blue-500 bg-white dark:bg-[#18181B]"
                      >
                        <option value="higher_better">Makin Tinggi Baik</option>
                        <option value="lower_better">Makin Rendah Baik</option>
                      </select>
                    </div>
                  </div>
                </>
              )}

              <div className="pt-4 flex justify-end gap-2 border-t border-slate-100 dark:border-gray-800 mt-4">
                <button type="button" onClick={() => setModalObj({...modalObj, isOpen: false})} className="px-4 py-2 border border-slate-300 dark:border-gray-600 text-slate-700 dark:text-gray-200 rounded-lg hover:bg-slate-50 dark:hover:bg-gray-800 text-sm font-semibold">Batal</button>
                <button type="submit" className="px-4 py-2 bg-blue-600 text-white rounded-lg hover:bg-blue-700 text-sm font-semibold flex items-center gap-2 shadow-sm">
                  <CheckCircle size={16} /> Simpan
                </button>
              </div>
            </form>
          </div>
        </div>
      )}

    </motion.div>
  );
};
